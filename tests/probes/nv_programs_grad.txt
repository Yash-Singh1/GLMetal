%% glsl txd_1d
#version 120
#extension GL_ARB_shader_texture_lod : require
uniform sampler1D tex6;
void main() {
  vec4 a = gl_TexCoord[0], b = gl_TexCoord[1], c = gl_TexCoord[2], i = gl_TexCoord[3];
  gl_FragColor = texture1DGradARB(tex6, a.x, b.x, c.x);
}
